name: Build Windows EXE

on:
  workflow_dispatch:
    inputs:
      release_version:
        description: "Optional version to publish, for example v0.4.2. Leave blank to build only."
        required: false
        type: string

permissions:
  contents: write

jobs:
  build:
    runs-on: windows-2022

    steps:
      - name: Download repository
        uses: actions/checkout@v6

      - name: Install Python 3.12
        uses: actions/setup-python@v6
        with:
          python-version: "3.12"
          cache: pip
          cache-dependency-path: requirements-build.txt

      - name: Install build dependencies
        shell: pwsh
        run: |
          python -m pip install --upgrade pip
          python -m pip install -r requirements-build.txt

      - name: Run tests
        shell: pwsh
        run: python -m pytest -q

      - name: Build standalone EXE
        shell: pwsh
        run: pyinstaller --noconfirm --clean BPSR-MIDI-Lite.spec

      - name: Prepare downloads
        shell: pwsh
        run: |
          New-Item -ItemType Directory -Force release | Out-Null
          Copy-Item dist/BPSR-MIDI-Lite.exe release/BPSR-MIDI-Lite.exe

          New-Item -ItemType Directory -Force portable/BPSR-MIDI-Lite | Out-Null
          Copy-Item dist/BPSR-MIDI-Lite.exe portable/BPSR-MIDI-Lite/BPSR-MIDI-Lite.exe
          Copy-Item README.md portable/BPSR-MIDI-Lite/README.md
          Copy-Item END_USER_GUIDE.md portable/BPSR-MIDI-Lite/END_USER_GUIDE.md
          Copy-Item LICENSE portable/BPSR-MIDI-Lite/LICENSE
          Copy-Item THIRD_PARTY_NOTICES.md portable/BPSR-MIDI-Lite/THIRD_PARTY_NOTICES.md
          New-Item -ItemType Directory -Force portable/BPSR-MIDI-Lite/MIDI | Out-Null
          Copy-Item MIDI/PUT_MIDI_FILES_HERE.txt portable/BPSR-MIDI-Lite/MIDI/PUT_MIDI_FILES_HERE.txt

          Compress-Archive -Path portable/BPSR-MIDI-Lite -DestinationPath release/BPSR-MIDI-Lite-Windows-x64.zip -Force

          $hashExe = (Get-FileHash release/BPSR-MIDI-Lite.exe -Algorithm SHA256).Hash.ToLower()
          $hashZip = (Get-FileHash release/BPSR-MIDI-Lite-Windows-x64.zip -Algorithm SHA256).Hash.ToLower()
          "$hashExe  BPSR-MIDI-Lite.exe" | Set-Content release/SHA256SUMS.txt
          "$hashZip  BPSR-MIDI-Lite-Windows-x64.zip" | Add-Content release/SHA256SUMS.txt

      - name: Upload build artifact
        uses: actions/upload-artifact@v6
        with:
          name: BPSR-MIDI-Lite-Windows
          path: release/*
          if-no-files-found: error
          retention-days: 30

      - name: Publish GitHub Release when a version was entered
        if: ${{ inputs.release_version != '' }}
        shell: pwsh
        env:
          GH_TOKEN: ${{ github.token }}
          RELEASE_VERSION: ${{ inputs.release_version }}
        run: |
          $tag = $env:RELEASE_VERSION.Trim()
          if (-not $tag.StartsWith("v")) { $tag = "v$tag" }

          gh release create $tag `
            release/BPSR-MIDI-Lite.exe `
            release/BPSR-MIDI-Lite-Windows-x64.zip `
            release/SHA256SUMS.txt `
            --target $env:GITHUB_SHA `
            --title "BPSR MIDI Lite $tag" `
            --generate-notes
